임시 검사 스크립트의 소스는 아래와 같다. 저장소 루트에서 실행한다. 외부 URL 접속이나 Markdown 렌더러 전체를 검증하는 도구는 아니다.

```python
from pathlib import Path
import re
r=Path.cwd(); bad=[]; total=0
for p in r.rglob('*.md'):
    if '.git' in p.parts: continue
    text=re.sub(r'```.*?```','',p.read_text(),flags=re.S)
    for target in re.findall(r'(?<!!)\[[^\]]*\]\(([^)]+)\)',text):
        if re.match(r'\w+://',target): continue
        path,_,anchor=target.partition('#'); q=(p.parent/path).resolve() if path else p
        total+=1
        if not q.exists(): bad.append(f'{p.relative_to(r)}: missing {target}')
        elif anchor and q.is_file():
            heads=re.findall(r'^#{1,6}\s+(.+?)\s*$',q.read_text(),re.M)
            slugs=[re.sub(r'[^\w\- ]','',h.lower()).replace(' ','-') for h in heads]
            if anchor not in slugs: bad.append(f'{p.relative_to(r)}: missing anchor {target}')
assert not bad, '\n'.join(bad)
source=(r/'docs/archive/PRD-v0.8.md').read_text()
aug=(r/'docs/prd/AUGMENTS.md').read_text()
ids=lambda s:re.findall(r'^\| `([^`]+)` \|',s,re.M)
old=source[source.index('**MVP 증강 19종**'):source.index('### 3.7')]
assert ids(aug)==ids(old) and len(ids(aug))==19
expected=['AGENTS','README','PRD','TASKS','PROGRESS','DESIGN','DECISIONS','TROUBLESHOOTING','ENGINEERING-VALIDATION','CHANGELOG']
assert all((r/f'{name}.md').is_file() for name in expected)
for p in (r/'docs/design').glob('*.md'):
    t=p.read_text(); assert all(s in t[:600] for s in ['승인된 미구현 설계','현재 구현','미정']),p
assert all(not (r/f'docs/{n}.md').exists() for n in ['PRD','TASKS','PROGRESS','DESIGN','DECISIONS'])
print(f'PASS: {total} local Markdown links and anchors')
print('PASS: 10 root documents; former duplicate entrypoints removed')
print('PASS: 19 augment IDs preserved in order; detailed designs separate approved/unimplemented, implemented, pending')

```

검증 결과: 상대 Markdown 링크·앵커 200개 검사 통과. 루트 기준 문서 10개와 이전 중복 진입점 제거 확인. 증강 19종 ID·순서 유지, 상세 설계의 승인된 미구현/현재 구현/미정 구분 확인. `git diff --check` 통과. 문서 책임·참조·협업 양식의 수동 대조도 완료했다.
